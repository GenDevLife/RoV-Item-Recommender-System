```python
def get_recommendation(self, hero_code, role, lane, ga_profile):
    # 1. ดึงข้อมูล Hero
    hero_data = self.repo.get_hero_data(hero_code)
    
    # 2. ใส่ role/lane ที่เลือก
    hero_data['selected_role'] = role
    hero_data['selected_lane'] = lane
    
    # 3. ดึง GA Settings ตาม Profile
    ga_settings = get_ga_settings(ga_profile)
    # Fast: {POP_SIZE: 30, MAX_GEN: 50, MUTATION_RATE: 0.3, ELITISM_COUNT: 3}
    
    # 4. สร้าง Evaluator
    evaluator = BuildEvaluator(hero_data, self.all_items)
    
    # 5. สร้าง GA Engine
    engine = GeneticEngine(evaluator, self.valid_items, ga_settings)
    
    # 6. รัน GA และรับผลลัพธ์
    build, score = engine.run()
    
    return build, score, hero_data
```

6.3 calculate_stats() - คำนวณ Stats รวม

```python
def calculate_stats(self, chromosome):
    # เริ่มจาก Base Stats ของ Hero
    stats = {
        "p_atk": self.hero.get('base_atk') or 100,   # 168 (Valhein)
        "p_def": self.hero.get('base_def') or 50,    # 89
        "max_hp": self.hero.get('base_hp') or 3000,  # 3420
        "m_power": 0.0,
        "cdr": 0.0,
        "aspd": 0.0,
        "crit_rate": 0.0,
        "move_speed": 350.0,  # Base move speed
        "p_pierce_percent": 0.0,
        "m_pierce_percent": 0.0
    }
    
    # บวก Stats จากไอเทมแต่ละชิ้น
    for item_id in chromosome:
        item = self.all_items.get(item_id)
        
        stats["p_atk"] += item.get("p_atk", 0)
        stats["m_power"] += item.get("m_power", 0)
        stats["p_def"] += item.get("p_def", 0)
        stats["max_hp"] += item.get("max_hp", 0)
        stats["cdr"] += item.get("cdr", 0)
        stats["aspd"] += item.get("aspd", 0)
        stats["crit_rate"] += item.get("crit_rate", 0)
        stats["move_speed"] += item.get("move_speed", 0)
        
        # Pierce เอาค่าสูงสุด (ไม่รวมกัน)
        stats["p_pierce_percent"] = max(stats["p_pierce_percent"], 
                                        item.get("p_pierce_percent", 0))
    
    return stats
```

6.4 get_fitness() - คำนวณคะแนน Fitness
────────────────────────────────────────

```python
def get_fitness(self, chromosome):
    score = 0.0
    
    # ───────────────────────────────────────
    # STEP 1: ตรวจสอบ Penalty จาก Passive ซ้ำ
    # ───────────────────────────────────────
    item_objects = [self.all_items[i] for i in chromosome if i in self.all_items]
    passive_penalty, conflicts = self.passive_manager.check_passive_conflicts(item_objects)
    score += passive_penalty  # -50 ต่อความขัดแย้ง 1 คู่
    
    # ───────────────────────────────────────
    # STEP 2: เช็ครองเท้าซ้ำ
    # ───────────────────────────────────────
    boots_count = sum(1 for item in item_objects 
                      if 'limit_one_boots' in item.get('restrictions', []))
    if boots_count > 1:
        score += PENALTIES['boots_limit'] * (boots_count - 1)
        # ถ้ามี 2 คู่รองเท้า: score += -100 * 1 = -100
    
    # ───────────────────────────────────────
    # STEP 3: คำนวณ Stats รวม
    # ───────────────────────────────────────
    stats = self.calculate_stats(chromosome)
    
    # ───────────────────────────────────────
    # STEP 4: บังคับ Stat Caps (ตามกฎเกม)
    # ───────────────────────────────────────
    effective_cdr = min(stats['cdr'], 0.40)      # CDR สูงสุด 40%
    effective_crit = min(stats['crit_rate'], 1.00)  # Crit สูงสุด 100%
    effective_aspd = min(stats['aspd'], 2.00)    # ASPD สูงสุด 200%
    
    # ───────────────────────────────────────
    # STEP 5: คำนวณคะแนนตาม Weights
    # ───────────────────────────────────────
    score += stats['p_atk'] * self.weights.get('p_atk', 0)
    # 673 × 0.15 = 100.95
    
    score += stats['m_power'] * self.weights.get('ap', 0)
    # 0 × 0.15 = 0
    
    score += stats['max_hp'] * self.weights.get('hp', 0)
    # 3420 × 0.001 = 3.42
    
    score += effective_cdr * self.weights.get('cdr', 0)
    # 0.10 × 0.20 = 0.02 (ถ้ามี 10% CDR)
    
    score += effective_aspd * self.weights.get('aspd', 0)
    # 0.35 × 0.50 = 0.175
    
    score += effective_crit * self.weights.get('crit', 0)
    # 0.50 × 0.80 = 0.40
    
    score += stats['p_pierce_percent'] * 100 * self.weights.get('p_pierce', 0)
    # 0.40 × 100 × 1.20 = 48.0
    
    return score
```

ตัวอย่างคะแนนรวม:
```
Fitness Score = 0 (เริ่มต้น)
              + 0 (ไม่มี passive ซ้ำ)
              + 0 (ไม่มีรองเท้าซ้ำ)
              + 100.95 (p_atk)
              + 0 (m_power)
              + 3.42 (HP)
              + 0.02 (CDR)
              + 0.175 (ASPD)
              + 0.40 (Crit)
              + 48.0 (Armor Pierce)
```

```python
def check_passive_conflicts(self, build_items):
    seen_groups = set()  # เก็บ passive group ที่เจอแล้ว
    conflicts = []
    total_penalty = 0.0
    
    for item in build_items:
        passives = item.get('passives', [])
        # เช่น passives = ['elemental_power'] สำหรับ Omni Arms
        
        for group in passives:
            if group in seen_groups:
                # พบ passive ซ้ำ!
                conflicts.append(f"Passive '{group}' ซ้ำ")
                total_penalty += -50.0  # หัก 50 คะแนน
            else:
                seen_groups.add(group)
    
    return total_penalty, conflicts
```

ตัวอย่าง:
```
Build มี: [Omni Arms, Frost Cape, War Boots]
- Omni Arms มี passive: ['elemental_power']
- Frost Cape มี passive: ['elemental_power']  ← ซ้ำ!
- War Boots มี passive: ['unique_movement']

ผลลัพธ์: penalty = -50, conflicts = ["Passive 'elemental_power' ซ้ำ"]
```

8.3 crossover() - ผสมพันธุ์ระหว่าง 2 Parents
────────────────────────────────────────────

```python
def crossover(self, parent1, parent2):
    # สุ่มจุดตัด (1 ถึง 5)
    point = random.randint(1, 5)  # สมมติได้ 3
    
    # ตัดแบ่ง
    child = parent1[:point] + parent2[point:]
    # parent1 = [A, B, C, D, E, F]
    # parent2 = [G, H, I, J, K, L]
    # point = 3
    # child = [A, B, C] + [J, K, L] = [A, B, C, J, K, L]
    
    # แก้ไอเทมซ้ำ
    return self.ensure_unique_items(child)
```

8.4 ensure_unique_items() - แก้ไอเทมซ้ำ
────────────────────────────────────────

```python
def ensure_unique_items(self, chromosome):
    seen = set()
    result = []
    
    for item_id in chromosome:
        if item_id not in seen:
            result.append(item_id)
            seen.add(item_id)
        else:
            # มีไอเทมซ้ำ → หาไอเทมใหม่
            available = [i for i in self.item_pool if i not in seen]
            if available:
                new_item = random.choice(available)
                result.append(new_item)
                seen.add(new_item)
    
    return result
```

ตัวอย่าง:
```
Input:  [1, 5, 5, 8, 12, 1]  # มี 5 และ 1 ซ้ำ
Output: [1, 5, 23, 8, 12, 45]  # แทนที่ด้วยไอเทมอื่น
```

8.5 mutate() - กลายพันธุ์
─────────────────────────

```python
def mutate(self, chromosome):
    if random.random() < self.mutation_rate:  # 30% โอกาส
        idx = random.randint(0, 5)  # สุ่มตำแหน่ง (0-5)
        available = [i for i in self.item_pool if i not in chromosome]
        if available:
            chromosome[idx] = random.choice(available)
    
    return chromosome
```

```
↓ mutate ที่ตำแหน่ง 2
After:  [1, 5, 88, 23, 45, 67]  # แทน 12 ด้วย 88
```

8.6 run() - รัน Genetic Algorithm
──────────────────────────────────

```python
def run(self):
    # ══════════════════════════════════════════════
    # STEP 1: สร้างประชากรเริ่มต้น
    # ══════════════════════════════════════════════
    population = [self.create_chromosome() for _ in range(self.pop_size)]
    # population = [ [..build1..], [..build2..], ..., [..build30..] ]
    
    best_solution = []
    best_fitness = -999999.0
    
    # ══════════════════════════════════════════════
    # STEP 2: วนลูป Generations
    # ══════════════════════════════════════════════
    for gen in range(self.max_gen):  # 0 to 49 (50 รอบ)
        
        # ─────────────────────────────────────────
        # 2.1 ประเมิน Fitness ของทุกตัว
        # ─────────────────────────────────────────
        scores = [(chrom, self.evaluator.get_fitness(chrom)) for chrom in population]
        # scores = [
        #     ([1,5,12,23,45,67], 152.97),
        #     ([2,8,15,30,42,55], 140.23),
        #     ...
        # ]
        
        # เรียงจากคะแนนสูงสุดไปต่ำสุด
        scores.sort(key=lambda x: x[1], reverse=True)
        
        # ─────────────────────────────────────────
        # 2.2 เก็บตัวที่ดีที่สุด (ตลอดทุก Generation)
        # ─────────────────────────────────────────
        if scores[0][1] > best_fitness:
            best_fitness = scores[0][1]
            best_solution = scores[0][0][:]  # Copy build
        
        # ─────────────────────────────────────────
        # 2.3 สร้าง Generation ถัดไป
        # ─────────────────────────────────────────
        
        # Elitism: เก็บ top 3 ไว้ก่อน (ไม่ผ่าน crossover/mutation)
        next_gen = [s[0][:] for s in scores[:self.elitism_count]]
        # next_gen = [top1_build, top2_build, top3_build]
        
        # สร้าง builds ใหม่จนครบ pop_size
        while len(next_gen) < self.pop_size:
            # เลือก parents จาก top 10
            parent1 = random.choice(scores[:10])[0]
            parent2 = random.choice(scores[:10])[0]
            
            # Crossover
            child = self.crossover(parent1[:], parent2[:])
            
            # Mutation
            child = self.mutate(child)
            
            next_gen.append(child)
        
        # อัพเดท population สำหรับรอบถัดไป
        population = next_gen
    
    # ══════════════════════════════════════════════
    # STEP 3: Return ผลลัพธ์ที่ดีที่สุด
    # ══════════════════════════════════════════════
    return best_solution, best_fitness
    # ([1, 5, 12, 23, 45, 67], 361.46)
```

Logic หลักของระบบ:

1. DATABASE LAYER (repository.py)
   - เก็บข้อมูล heroes, items, passives, restrictions
   - ใช้ SQLite สำหรับ query ข้อมูล

2. EVALUATION LAYER (evaluator.py)
   - calculate_stats(): รวม stats ของ hero + items
   - get_fitness(): ให้คะแนน build (Weighted Sum - Penalties)
   - Weights มาจาก calibration ด้วย Ridge Regression

3. OPTIMIZATION LAYER (ga_engine.py)
   - Genetic Algorithm หา build ที่ดีที่สุด
   - Operators: Selection, Crossover, Mutation, Elitism
   - 3 Profiles: Fast (50 gen), Medium (100 gen), Expert (150 gen)

4. APPLICATION LAYER (main.py)
   - CLI interface สำหรับ user interaction
   - 2 modes: Single Hero, All Heroes (batch)
   - Output: Console หรือ CSV file

5. CALIBRATION LAYER (calibrate.py)
   - เรียนรู้ weights จาก training data
   - ใช้ Ridge Regression + StandardScaler
   - Output: learned_weights.json
